example_workflow.py 
from automaton import Workflow, WebApp

# Define your automation
async def process_order(customer_data):
    email = await send_confirmation(customer_data)
    inventory = await update_stock()
    return {"status": "success"}

# Host your web app
app = WebApp(backend=Workflow.from_function(process_order))
app.deploy()  # That's it. No servers needed.
